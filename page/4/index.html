  // 判断调用call的函数（也就是this）是否是函数
  if(typeof this === 'function') {
    throw new TypeError('Error')
  }
  context = context || window
  context.fn = this
  const args = [...argumments].slice(1)
  const result = context.fn(args)
  delete context.fn
  return result
}
